cmake_minimum_required(VERSION 3.12.0)
project(gmsf LANGUAGES C)

include(CheckCSourceCompiles)
include(CMakePushCheckState)
include(GNUInstallDirs)
include(GenerateExportHeader)
include(FindPackageHandleStandardArgs)

set(CMAKE_C_STANDARD 90)
add_compile_options(-Wno-deprecated-declarations)
add_compile_definitions(GLIB_DISABLE_DEPRECATION_WARNINGS)

option(BUILD_SHARED_LIBS "Build as a shared library" TRUE)
option(BUILD_EXAMPLES "Build examples" TRUE)

find_package(PkgConfig REQUIRED)
set(PC_DEPS_LIST glib-2.0 gobject-2.0 gio-2.0 gdk-pixbuf-2.0)
pkg_check_modules(PC_DEPS REQUIRED IMPORTED_TARGET ${PC_DEPS_LIST})
find_path(LIBPE_INCLUDE_DIR NAMES libpe/pe.h)
find_library(LIBPE_LIBRARY NAMES pe libpe)
find_package_handle_standard_args(LibPE REQUIRED_VARS LIBPE_INCLUDE_DIR LIBPE_LIBRARY)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/include/gmsf.h" GMSFHDR)
string(REGEX MATCH "GMSF_VERSION_MAJOR ([0-9]*)" _ ${GMSFHDR})
set(VERSION_MAJOR ${CMAKE_MATCH_1})
string(REGEX MATCH "GMSF_VERSION_MINOR ([0-9]*)" _ ${GMSFHDR})
set(VERSION_MINOR ${CMAKE_MATCH_1})

file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
add_library(libgmsf ${SOURCE_FILES})
add_library(gmsf::libgmsf ALIAS libgmsf)
generate_export_header(libgmsf BASE_NAME "gmsf" EXPORT_FILE_NAME "gmsf-export.h")
set_property(TARGET libgmsf PROPERTY OUTPUT_NAME "gmsf")
set_property(TARGET libgmsf PROPERTY C_VISIBILITY_PRESET "hidden")
set_property(TARGET libgmsf PROPERTY POSITION_INDEPENDENT_CODE ON)
set_target_properties(libgmsf PROPERTIES VERSION ${VERSION_MAJOR}.${VERSION_MINOR} SOVERSION ${VERSION_MAJOR})
target_include_directories(libgmsf PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_include_directories(libgmsf PUBLIC "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_include_directories(libgmsf PUBLIC "${CMAKE_BINARY_DIR}")
target_link_libraries(libgmsf PUBLIC PkgConfig::PC_DEPS)
target_include_directories(libgmsf PUBLIC ${LIBPE_INCLUDE_DIR})
target_link_libraries(libgmsf PUBLIC ${LIBPE_LIBRARY})

string(JOIN " " PC_DEPS_PRIVATE ${PC_DEPS_LIST})
configure_file(gmsf.pc.in gmsf.pc @ONLY)

install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/include/gmsf.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/gmsf-export.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/gmsf.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
install(TARGETS libgmsf ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

if(BUILD_EXAMPLES)
	file(GLOB_RECURSE EXAMPLE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/examples/*.c")
	foreach(EXAMPLE_FILE ${EXAMPLE_FILES})
		get_filename_component(EXAMPLE_TARGET ${EXAMPLE_FILE} NAME_WE)
        add_executable(${EXAMPLE_TARGET} ${EXAMPLE_FILE})
        target_link_libraries(${EXAMPLE_TARGET} PRIVATE gmsf::libgmsf)
	endforeach()
endif()
